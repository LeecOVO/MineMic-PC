cmake_minimum_required(VERSION 3.24)
project(MineMicPC VERSION 0.0.1 LANGUAGES CXX)
set(MINEMIC_PRERELEASE "ea.1")
set(MINEMIC_BUILD_NUMBER 4)
set(MINEMIC_VERSION "${PROJECT_VERSION}-${MINEMIC_PRERELEASE}")
configure_file(resources/Version.h.in generated/Version.h @ONLY)
configure_file(resources/windows.rc.in generated/windows.rc @ONLY)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MINEMIC_BUILD_APP "Build the Qt desktop application" ON)
option(MINEMIC_SINGLE_FILE "Require static Qt and a static MSVC runtime" OFF)
option(MINEMIC_DEVICE_TESTS "Run an audio-device probe (opens output, plays silence)" OFF)
include(CTest)

if(MSVC)
    add_compile_options(/W4 /utf-8 /permissive-)
    if(MINEMIC_SINGLE_FILE)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(WIN32 AND NOT MSVC AND MINEMIC_BUILD_APP)
    message(FATAL_ERROR "The Windows WASAPI backend currently requires the MSVC toolchain (VS 2022).")
endif()

add_library(minemic_core STATIC src/core/Packet.cpp src/core/AudioBuffer.cpp)
target_include_directories(minemic_core PUBLIC src)

if(BUILD_TESTING)
    add_executable(minemic_core_tests tests/CoreTests.cpp)
    target_link_libraries(minemic_core_tests PRIVATE minemic_core)
    add_test(NAME core COMMAND minemic_core_tests)
endif()

if(MINEMIC_BUILD_APP)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network)
    qt_standard_project_setup()
    if(MINEMIC_SINGLE_FILE)
        get_target_property(_qt_type Qt6::Core TYPE)
        if(NOT _qt_type STREQUAL "STATIC_LIBRARY")
            message(FATAL_ERROR "MINEMIC_SINGLE_FILE requires a static Qt build. See docs/BUILDING.md.")
        endif()
        if(NOT WIN32)
            message(FATAL_ERROR "The single-file packaging recipe targets Windows only.")
        endif()
    endif()

    add_library(minemic_network STATIC src/network/Receiver.cpp src/network/Receiver.h src/network/LanAdvertiser.cpp src/network/LanAdvertiser.h)
    target_link_libraries(minemic_network PUBLIC minemic_core Qt6::Core Qt6::Network)

    add_library(minemic_audio STATIC src/audio/AudioOutput.h)
    set_target_properties(minemic_audio PROPERTIES AUTOMOC OFF AUTOUIC OFF)
    target_link_libraries(minemic_audio PUBLIC minemic_core Qt6::Core)

    qt_add_executable(MineMic WIN32 MACOSX_BUNDLE
        src/main.cpp src/ui/MainWindow.cpp src/ui/MainWindow.h
        src/audio/AudioOutput.h)
    target_link_libraries(MineMic PRIVATE minemic_network minemic_audio Qt6::Widgets)
    target_include_directories(MineMic PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(WIN32)
        target_sources(minemic_audio PRIVATE src/audio/WasapiOutput.cpp)
        target_sources(MineMic PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/windows.rc")
        target_compile_definitions(minemic_audio PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
        target_link_libraries(minemic_audio PRIVATE ole32 uuid avrt propsys)
        if(MINEMIC_SINGLE_FILE)
            qt_import_plugins(MineMic NO_DEFAULT INCLUDE Qt6::QWindowsIntegrationPlugin)
            foreach(_style QModernWindowsStylePlugin QWindowsVistaStylePlugin)
                if(TARGET Qt6::${_style})
                    qt_import_plugins(MineMic INCLUDE Qt6::${_style})
                    break()
                endif()
            endforeach()
        endif()
    elseif(APPLE)
        target_sources(minemic_audio PRIVATE src/audio/CoreAudioOutput.cpp)
        target_link_libraries(minemic_audio PRIVATE "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation")
        set_target_properties(MineMic PROPERTIES
            MACOSX_BUNDLE_BUNDLE_NAME "MineMic"
            MACOSX_BUNDLE_GUI_IDENTIFIER "com.aiaixyz.minemic.pc"
            MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
            MACOSX_BUNDLE_BUNDLE_VERSION "${MINEMIC_BUILD_NUMBER}")
    else()
        target_sources(minemic_audio PRIVATE src/audio/UnsupportedOutput.cpp)
    endif()
    install(TARGETS MineMic BUNDLE DESTINATION . RUNTIME DESTINATION .)

    if(BUILD_TESTING)
        add_executable(minemic_audio_probe tests/AudioProbe.cpp)
        target_link_libraries(minemic_audio_probe PRIVATE minemic_audio)
        if(MINEMIC_DEVICE_TESTS)
            add_test(NAME audio_device COMMAND minemic_audio_probe)
            set_tests_properties(audio_device PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 15)
        endif()
        add_executable(minemic_network_tests tests/NetworkTests.cpp)
        target_link_libraries(minemic_network_tests PRIVATE minemic_network)
        add_test(NAME network COMMAND minemic_network_tests)
        find_package(Python3 QUIET COMPONENTS Interpreter)
        if(Python3_Interpreter_FOUND)
            add_test(NAME python_interop COMMAND minemic_network_tests --python-sender "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/send_test_audio.py")
        endif()
        add_test(NAME ui_smoke COMMAND MineMic --smoke-test)
        set_tests_properties(ui_smoke PROPERTIES TIMEOUT 15)
        if(NOT WIN32)
            set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        endif()
    endif()
endif()
